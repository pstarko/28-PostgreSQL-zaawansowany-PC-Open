explain analyze
select
	last_name 
from employees
where
	last_name ~*'K'
	
	
explain analyze
select
	last_name 
from employees
where
	lower(last_name) like '%k%'
	
select
	last_name 
from employees
where
	last_name ilike 'k%'
	
explain analyze
select
	last_name 
from employees
where
	last_name ~~'K%' -- = last_name like 'K%'
	
select
	last_name 
from employees
where
	last_name !~~'K%'
	
--szukanie ciągów z litery "k" case-sensitve
select
	last_name 
from employees
where
	last_name ~'k' --> = last_name like '%k%'

--szukanie ciągów z litery "k" case-insensitve
select
	last_name 
from employees
where
	last_name ~*'K'

select
	*
from employees
where last_name similar to '\mK%o\M'

select
	concat(first_name || ' ' || last_name)
from employees
where concat(first_name || ' ' || last_name) similar to '%\mY%g\M'

select
	concat(first_name || ' ' || last_name
from employees
where concat(first_name || last_name) similar to '%\mY%g\M'

	last_name,
	substring(last_name similar '#mK%o{2}' escape '#') AS match
