/*
1. Opisz za pomocą REGEXP_LIKE

select
	REGEXP_LIKE('12345678901', '^[0-9]{11}$')


/*
1. Opisz za pomocą REGEXP_LIKE

b. Kod w formacie (2litery, 3 liczby, 1 litera) - tylko wielkie litery np., AG234D

d. Adres e-mail w formacie (imię.nazwa@domena.com)


*/

SELECT REGEXP_LIKE('AG234D', '^[A-Z]{2}[0-9]{3}[A-Z]{1}$')


SELECT REGEXP_LIKE('g.kowalski@domena.com', '^[A-Za-z]+\.[A-Za-z]+@[A-Za-z]{2,}\.com$');

/*
1. Opisz za pomocą REGEXP_LIKE

g. Numer telefonu w formacie: (0-61) 878 23 98 (bez spacji)

2. Pokaż nazwiska pracowników (wielkość liter nie ma znaczenia):
a. które kończą się na „i” i nie zaczynają się na „d”

*/

SELECT REGEXP_LIKE('(0-61)87823 98', '^\(0-[0-9]{2}\)[0-9]{3}[0-9]{2} [0-9]{2}$');


SELECT last_name
FROM Employees
WHERE REGEXP_LIKE(last_name, 'i$', 'i') 
  AND NOT REGEXP_LIKE(last_name, '^d', 'i');



SELECT last_name
FROM employees
WHERE REGEXP_LIKE(last_name, '^[^d].*i$', 'i'); --^[^d] - pierwszy znak nie jest literą d
