--SETOF composite_type
-- Najpierw definiujemy typ złożony
CREATE TYPE employee_info AS (
    employee_id numeric,
    full_name text,
    job_title varchar(50),
    rate numeric
);

-- Następnie tworzymy funkcję zwracającą zbiór tego typu
CREATE OR REPLACE FUNCTION get_employees_info(p_dept_id numeric)
RETURNS SETOF employee_info AS $$
DECLARE
    emp_record employee_info;
BEGIN
    -- Możemy wykonać złożoną logikę przed zwróceniem wyników
    -- np. walidację, obliczenia, itp.    
    FOR emp_record IN (SELECT 
            employee_id,
            first_name || ' ' || last_name,
            job_id,
            salary
        FROM employees
        WHERE department_id = p_dept_id) LOOP
        RETURN NEXT emp_record;
    END LOOP;
    
    RETURN;
END;
$$ LANGUAGE plpgsql;


select * from get_employees_info(2)

--SETOF record
CREATE OR REPLACE FUNCTION get_complex_employee_data(p_rate numeric)
RETURNS SETOF record AS $$
DECLARE
    temp_table_name text;
    query text;
BEGIN
    -- Tworzymy tymczasową tabelę
    temp_table_name := 'temp_emp_' || md5(random()::text);
    
    -- Poprawiona składnia - USING jest częścią instrukcji EXECUTE
    EXECUTE format('CREATE TEMP TABLE %I AS
        SELECT 
            departmentname,
            COUNT(*) AS employee_count,
            AVG(rate) AS avg_rate
        FROM emp
        WHERE rate > $1
        GROUP BY departmentname', temp_table_name)
    USING p_rate;
    
    -- Możemy wykonać dodatkowe operacje na tymczasowej tabeli
    EXECUTE format('UPDATE %I SET avg_rate = ROUND(avg_rate, 2)', temp_table_name);
    
    -- Zwracamy wyniki
    RETURN QUERY EXECUTE format('SELECT * FROM %I', temp_table_name);
    
    -- Usuwamy tymczasową tabelę
    EXECUTE format('DROP TABLE %I', temp_table_name);
END;
$$ LANGUAGE plpgsql;


SELECT * FROM get_complex_employee_data(15) AS 
(dept_name varchar, emp_count bigint, average_rate numeric);

create or replace function fun6(var_salary numeric)
	returns table (
		id employees.employee_id%TYPE,
		ln varchar
	)
language plpgsql
as $$
declare
	var_record record;
begin 
	for var_record in (
		select employee_id, last_name
		from employees
		where salary>var_salary
	) loop 
		id:=var_record.employee_id;
		ln:=var_record.last_name;
	return next;
	end loop;	
end $$

/*
IMMUTABLE wskazuje, że funkcja nie może modyfikować bazy danych i zawsze zwraca ten sam wynik, 
gdy otrzymuje te same wartości argumentów; 
oznacza to, że nie wykonuje wyszukiwania w bazie danych ani w inny sposób nie wykorzystuje informacji, 
które nie są bezpośrednio obecne na liście argumentów. 
Jeśli ta opcja jest podana, każde wywołanie funkcji ze stałymi argumentami może zostać natychmiast zastąpione 
wartością funkcji.

IMMUTABLE musi być czystą funkcją, której wyniki zależą tylko od jej danych wejściowych. 
Jest to bardzo surowy wymóg; 

nie mogą one wywoływać innych niezmiennych funkcji, 
nie mogą uzyskiwać dostępu do tabel(DML), 
nie mogą uzyskiwać dostępu do wartości właściwości konfiguracji itp.
*/
--stwórz najpierw bez IMMUTABLE i pokaże EXPLAIN, następnie dodaj IMMUTABLE i zobacz różnicę
create or replace function get_name(dept_id numeric)
returns table (first_name varchar, last_name varchar, deptartment_id int)
immutable
as $$
	select 
		first_name,
		last_name,
		department_id
		from employees
	where department_id = dept_id;
$$ language sql;
