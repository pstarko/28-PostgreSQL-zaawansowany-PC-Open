--Wyzwalacze

/*
PostgreSQL udostępnia dwa główne typy wyzwalaczy:

	Wyzwalacze na poziomie wiersza
	Wyzwalacze na poziomie instrukcji.
	
Różnice między tymi dwoma rodzajami polegają na tym, ile razy wyzwalacz jest wywoływany iw jakim czasie.

Na przykład, jeśli wydasz instrukcję UPDATE modyfikującą 20 wierszy, 
wyzwalacz na poziomie wiersza zostanie wywołany 20 razy, 
a wyzwalacz na poziomie instrukcji zostanie wywołany 1 raz.

Możesz określić, czy wyzwalacz jest wywoływany przed, czy po zdarzeniu. 

Jeśli wyzwalacz zostanie wywołany przed zdarzeniem, może pominąć operację dla bieżącego wiersza, 
a nawet zmienić aktualizowany lub wstawiany wiersz. 
W przypadku wywołania wyzwalacza po zdarzeniu wszystkie zmiany są dostępne dla wyzwalacza.


--Cechy charakterystyczne

Wyzwalacze PostgreSQL a standardowe wyzwalacze SQL
Mimo że PostgreSQL implementuje standard SQL, wyzwalacze w PostgreSQL mają pewne specyficzne cechy:

	PostgreSQL uruchamia wyzwalacz dla zdarzenia TRUNCATE.
	PostgreSQL umożliwia zdefiniowanie wyzwalacza na poziomie instrukcji w widokach.
	PostgreSQL wymaga zdefiniowania funkcji zdefiniowanej przez użytkownika jako akcji wyzwalacza, 
		podczas gdy standard SQL pozwala na użycie dowolnych poleceń SQL.


--Tworzenie
Aby utworzyć nowy wyzwalacz w PostgreSQL, wykonaj następujące kroki:

- Najpierw utwórz funkcję wyzwalacza za pomocą instrukcji CREATE FUNCTION.
- Po drugie, powiąż funkcję wyzwalacza z tabelą, używając instrukcji CREATE TRIGGER.
*/

--Syntax
CREATE TRIGGER trigger_name 
   {BEFORE | AFTER} { event }
   ON table_name
   [FOR [EACH] { ROW | STATEMENT }]
       EXECUTE PROCEDURE trigger_function
	   
	   
--przykład
DROP TABLE IF EXISTS emp;

--1
CREATE TABLE emp(
   id INT GENERATED ALWAYS AS IDENTITY,
   first_name VARCHAR(40) NOT NULL,
   last_name VARCHAR(40) NOT NULL,
   PRIMARY KEY(id)
);

--2
CREATE TABLE emp_audits (
   id INT GENERATED ALWAYS AS IDENTITY,
   employee_id INT NOT NULL,
   last_name VARCHAR(40) NOT NULL,
   changed_on TIMESTAMP(6) NOT NULL
);

--3
CREATE OR REPLACE FUNCTION log_last_name_changes()
  RETURNS TRIGGER 
  LANGUAGE PLPGSQL
  AS
$$
BEGIN
	IF NEW.last_name <> OLD.last_name THEN
		 INSERT INTO emp_audits(employee_id,last_name,changed_on)
		 VALUES(OLD.id,OLD.last_name,now());
	END IF;

	RETURN NEW;
END;
$$

--4
CREATE TRIGGER last_name_changes
  BEFORE UPDATE
  ON emp
  FOR EACH ROW
  EXECUTE PROCEDURE log_last_name_changes();

INSERT INTO emp (first_name, last_name)

---FOR STATEMENT
drop table test;
--1

CREATE TABLE test (
    num integer NOT NULL,
    text   character varying(50)
);


--2
CREATE OR REPLACE FUNCTION tr_test_for_each_statement()
   RETURNS trigger
   LANGUAGE plpgsql
AS
$$
DECLARE
    x_rec record;
BEGIN
    raise notice '=operacja: % =', TG_OP;
    IF (TG_OP = 'UPDATE' OR TG_OP = 'DELETE') THEN
        FOR x_rec IN SELECT * FROM old_table LOOP
            raise notice 'OLD: %', x_rec;
        END loop;
    END IF;
    IF (TG_OP = 'INSERT' OR TG_OP = 'UPDATE') THEN
        FOR x_rec IN SELECT * FROM new_table LOOP
            raise notice 'NEW: %', x_rec;
        END loop;
    END IF;

    RETURN NULL;
END;
$$;

--3
CREATE or replace TRIGGER tr_insert
   AFTER INSERT ON test
   REFERENCING NEW TABLE AS new_table
   FOR EACH STATEMENT
   EXECUTE PROCEDURE tr_test_for_each_statement();

CREATE or replace  TRIGGER tr_update
   AFTER UPDATE ON test
   REFERENCING NEW TABLE AS new_table OLD TABLE AS old_table
   FOR EACH STATEMENT
   EXECUTE PROCEDURE tr_test_for_each_statement();

CREATE or replace TRIGGER tr_delete
   AFTER DELETE ON test
   REFERENCING OLD TABLE AS old_table
   FOR EACH STATEMENT
   EXECUTE PROCEDURE tr_test_for_each_statement();
   
INSERT INTO test(num, text) VALUES (1, 'a');
/*
=operacja: INSERT =
NEW: (1,a)
*/

INSERT INTO test(num, text) VALUES (2, 'b'),  (3, 'b');
/*
=operacja: INSERT =
NEW: (2,b)
NEW: (3,b)
*/

UPDATE test SET num = num + 1 WHERE text = 'a';
/*
=operacja: UPDATE =
OLD: (1,a)
NEW: (2,a))
*/

UPDATE test SET num = num + 10 WHERE text = 'b';
/*
=operacja: UPDATE =
OLD: (2,b)
OLD: (3,b)
NEW: (12,b)
NEW: (13,b)
*/

DELETE FROM test
/*
=operacja: DELETE =
OLD: (2,a)
OLD: (12,b)
OLD: (13,b)
*/



--Zmiana wartości kolumny Triggerem
create table t1 (id int, a int, b int);

-- Funkcja wyzwalacza
CREATE OR REPLACE FUNCTION aktualizuj_kolumne_a()
RETURNS TRIGGER AS $$
BEGIN
    -- Dla INSERT - OLD nie istnieje
    IF TG_OP = 'INSERT' THEN
        NEW.b := COALESCE(NEW.b, 0) + COALESCE(NEW.a, 0);
    
    -- Dla UPDATE - sprawdź czy kolumna A się zmieniła
    ELSIF TG_OP = 'UPDATE' THEN
        NEW.b := COALESCE(NEW.b, 0) + COALESCE(NEW.a, 0);
    END IF;
    
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

-- Wyzwalacz
CREATE TRIGGER trigger_aktualizuj_a
    BEFORE update or insert ON t1
    FOR EACH ROW
    EXECUTE FUNCTION aktualizuj_kolumne_a();

truncate table t1;

insert into t1 values (2,5,10);

select * from t1;

update t1 set a = 3 where id = 2;
