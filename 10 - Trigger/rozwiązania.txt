create table Produkty 
	(idProduktu INT GENERATED ALWAYS AS identity primary key, 
	idMagazynu int,
	nazwaProduktu varchar(50), 
	cenaZakupu int )
	
	
create table Magazyn 
	(idMagazynu int primary key,
	Miasto varchar(50) )
	
alter table Produkty add foreign key (idMagazynu) 
references Magazyn (idMagazynu)


CREATE OR REPLACE PROCEDURE dodaj_produkt_i_magazyn(
    p_miasto VARCHAR(50),
    p_nazwa_produktu VARCHAR(50),
    p_cena_zakupu INT
)
LANGUAGE plpgsql
AS $$
DECLARE
    v_id_magazynu INT;
BEGIN
    -- Sprawdzenie, czy magazyn w danym mieście już istnieje
    SELECT idMagazynu INTO v_id_magazynu
    FROM Magazyn
    WHERE Miasto = p_miasto;
    
    -- Jeśli magazyn nie istnieje, dodaj nowy
    IF v_id_magazynu IS NULL THEN
        -- Znajdź następny dostępny ID dla magazynu
        SELECT COALESCE(MAX(idMagazynu), 0) + 1 INTO v_id_magazynu
        FROM Magazyn;
        
        -- Dodaj nowy magazyn
        INSERT INTO Magazyn (idMagazynu, Miasto)
        VALUES (v_id_magazynu, p_miasto);
    END IF;
    
	
    -- Dodaj produkt do istniejącego lub nowo utworzonego magazynu
    INSERT INTO Produkty (idMagazynu, nazwaProduktu, cenaZakupu)
    VALUES (v_id_magazynu, p_nazwa_produktu, p_cena_zakupu);
            
    RAISE NOTICE 'Dodano produkt % do magazynu w mieście %', p_nazwa_produktu, p_miasto;
EXCEPTION
    WHEN OTHERS THEN
        ROLLBACK;
        RAISE EXCEPTION 'Wystąpił błąd: %', SQLERRM;
END;
$$;

CREATE OR REPLACE FUNCTION sprawdz_cene_produktu()
RETURNS TRIGGER AS $$
BEGIN
    IF NEW.cenaZakupu <= 0 THEN
        RAISE EXCEPTION 'Nie można dodać produktu z ceną poniżej zera. Podana cena: %', NEW.cenaZakupu;
    END IF;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_sprawdz_cene_produktu
BEFORE INSERT OR UPDATE ON Produkty
FOR EACH ROW
EXECUTE FUNCTION sprawdz_cene_produktu();

-- Tworzenie tabeli logów
CREATE TABLE LogiZmianCen (
    id INT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    idProduktu INT NOT NULL,
    idMagazynu INT NOT NULL,
    staraCena INT,
    nowaCena INT NOT NULL,
    dataZmiany TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    typOperacji VARCHAR(10) NOT NULL
);

-- Tworzenie funkcji dla triggera
CREATE OR REPLACE FUNCTION log_zmian_cen()
RETURNS TRIGGER AS $$
BEGIN
    -- Dla operacji INSERT
    IF TG_OP = 'INSERT' THEN
        -- Dla nowych produktów zawsze zapisujemy log
		raise notice '% ', TG_OP;
        INSERT INTO LogiZmianCen (idProduktu, idMagazynu, staraCena, nowaCena, typOperacji)
        VALUES (NEW.idProduktu, NEW.idMagazynu, NULL, NEW.cenaZakupu, 'INSERT');
        
    -- Dla operacji UPDATE
    ELSIF TG_OP = 'UPDATE' THEN
        -- Sprawdzamy czy cena się zmieniła i czy nowa cena jest wyższa od starej
		raise notice '% ', TG_OP;
        IF NEW.cenaZakupu != OLD.cenaZakupu AND NEW.cenaZakupu > OLD.cenaZakupu THEN
            INSERT INTO LogiZmianCen (idProduktu, idMagazynu, staraCena, nowaCena, typOperacji)
            VALUES (NEW.idProduktu, NEW.idMagazynu, OLD.cenaZakupu, NEW.cenaZakupu, 'UPDATE');
        END IF;
    END IF;
    
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

-- Tworzenie triggera
CREATE TRIGGER trg_log_zmian_cen
AFTER INSERT OR UPDATE ON Produkty
FOR EACH ROW
EXECUTE FUNCTION log_zmian_cen();
